cmake_minimum_required(VERSION 3.16)
project(GutCPPUrlParity LANGUAGES CXX)
add_library(gutcpp_url INTERFACE)
target_include_directories(gutcpp_url INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(gutcpp_url INTERFACE cxx_std_17)
enable_testing()
add_executable(gutcpp_url_test tests/url_test.cpp)
target_link_libraries(gutcpp_url_test PRIVATE gutcpp_url)
add_test(NAME gutcpp_url_test COMMAND gutcpp_url_test)
