cmake_minimum_required(VERSION 3.16)
project(gutcpp_storage_path LANGUAGES CXX)
add_library(gutcpp_storage_path INTERFACE)
target_include_directories(gutcpp_storage_path INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(gutcpp_storage_path INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
 add_executable(storage_path_test tests/storage_path_test.cpp)
 target_link_libraries(storage_path_test PRIVATE gutcpp_storage_path)
 if(MSVC)
  target_compile_options(storage_path_test PRIVATE /W4 /WX)
 else()
  target_compile_options(storage_path_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
 add_test(NAME storage_path_test COMMAND storage_path_test)
endif()
