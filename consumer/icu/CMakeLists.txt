cmake_minimum_required(VERSION 3.16)
project(GutCPPInstalledUnicodeConsumer LANGUAGES CXX)
find_package(GutCPP 0.1 CONFIG REQUIRED)
add_executable(gutcpp_installed_icu_consumer main.cpp)
target_link_libraries(gutcpp_installed_icu_consumer PRIVATE gutcpp::icu)
if(MSVC)
  target_compile_options(gutcpp_installed_icu_consumer PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(gutcpp_installed_icu_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
