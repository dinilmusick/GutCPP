cmake_minimum_required(VERSION 3.16)
project(GutCPPInstalledConsumer LANGUAGES CXX)
include(CTest)
find_package(GutCPP 0.1 CONFIG REQUIRED)
add_executable(gutcpp_consumer main.cpp)
target_link_libraries(gutcpp_consumer PRIVATE gutcpp::gutcpp)
if(MSVC)
  target_compile_options(gutcpp_consumer PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(gutcpp_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME installed_consumer COMMAND gutcpp_consumer)
